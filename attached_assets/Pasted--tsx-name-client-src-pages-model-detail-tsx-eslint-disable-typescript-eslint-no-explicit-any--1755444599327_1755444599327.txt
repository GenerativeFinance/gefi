/**
 * Try to resolve an ID param from multiple possible routes or from pathname.
 * Historically the app used several patterns (/marketplace/:id, /model/:id, /models/:id).
 * This helper attempts the common patterns and falls back to parsing location.pathname.
 */
function resolveIdFromPathname(): string | null {
  if (typeof window === "undefined") return null;
  const path = window.location.pathname || "";
  const m = path.match(/(?:\/marketplace|\/model|\/models)\/([^\/?#]+)/i);
  if (m && m[1]) return decodeURIComponent(m[1]);
  return null;
}

    data: model,
    isLoading,
    error,
    refetch,

    ["/api/ai-models", idParam],
    async () => {
      if (!idParam) throw Object.assign(new Error("No model id provided"), { status: 400 });
      const resp = await apiRequest(`/api/ai-models/${idParam}`, "GET");

      // If helper returned a Response-like object, check status & parse JSON
      if (resp && typeof (resp as any).status === "number" && typeof (resp as any).json === "function") {
        if (!(resp as any).ok) {
          // Try to read body message for better UX
          let body;
          try {
            body = await (resp as any).json();
          } catch (e) {
            // ignore parse errors
          }
          const msg = body?.message || `Failed to fetch model (status ${resp.status})`;
          const err: any = new Error(msg);
          err.status = resp.status;
          throw err;
        }
        return (resp as any).json();
      }

      // If helper returned parsed JSON already
      return resp;
    },
    {
      enabled: !!idParam,
      retry: false,
      keepPreviousData: true,
      onError: (err: any) => {
        // show toast for non-404 errors
        if (err?.status && Number(err.status) === 404) {
          // let the "Model Not Found" UI handle this — no toast required
          return;
        }
        toast({
          title: "Failed to load model",
          description: String(err?.message || "Unable to load model details"),
          variant: "destructive",
        });
        // also log server error for debugging
        // eslint-disable-next-line no-console
        console.error("ModelDetail fetch error:", err);
      },
    }
  );

  // Subscribe mutation (keeps previous behavior but defensive about response shape)
  const subscribeMutation = useMutation({
    mutationFn: async () => {
      if (!idParam) throw new Error("No model id provided");
      const resp = await apiRequest(`/api/ai-models/${idParam}/subscribe`, "POST", {});
      if (resp && typeof (resp as any).json === "function") return (resp as any).json();
      return resp;
    },
    onSuccess: (data: any) => {
      if (data?.checkoutUrl) {
        // server wants us to redirect to Stripe Checkout
        window.location.href = data.checkoutUrl;
        return;
      }
      toast({
        title: "Successfully subscribed",
        description: "You now have access to this AI model.",
      });
      queryClient.invalidateQueries({ queryKey: ["/api/ai-models"] });
      queryClient.invalidateQueries({ queryKey: ["/api/portfolio/ai-models"] });
    },
    onError: (err: any) => {
      if (isUnauthorizedError(err)) {
        toast({
          title: "Unauthorized",
          description: "You need to be logged in to subscribe.",
          variant: "destructive",
        });
        return;
      }
      toast({
        title: "Subscription failed",
        description: String(err?.message || "Please try again later."),
        variant: "destructive",
      });
    },
  });

  useEffect(() => {
    if (error) {
      // log for developers; toast handled in onError above
      // eslint-disable-next-line no-console
      console.error("Error loading model:", error);
    }
  }, [error]);

  const handleRetry = async () => {
    try {
      await refetch();
    } catch (e) {
      toast({
        title: "Retry failed",
        description: "Unable to reload the model. Please try again later.",
        variant: "destructive",
      });

  // Defensive UI states
  if (!idParam) {

              <Button variant="outline" onClick={() => (window.location.href = "/marketplace")}>
                Browse Marketplace
              </Button>
              <Button onClick={() => (window.location.href = "/ai-models")}>All AI Models</Button>
            </div>
          </div>
        </div>
      </Layout>
    );
  }

  if (isLoading && !model) {
    return (
      <Layout>
        <div className="container mx-auto p-6 max-w-7xl">
          <div className="text-center">
            <div className="animate-spin w-12 h-12 border-4 border-blue-600 border-t-transparent rounded-full mx-auto mb-4"></div>
            <p className="text-gray-600 dark:text-gray-400">Loading model...</p>
          </div>
        </div>
      </Layout>
    );
  }

  // If there was an error and no cached model, show actionable error UI
  if (error && !model) {
    const message = (error as any)?.message || "Unable to load the model details. Please try again later.";

        <div className="container mx-auto p-6 max-w-7xl">

            <h1 className="text-2xl font-bold text-red-600 mb-4">Error Loading Model</h1>
            <p className="text-gray-600 dark:text-gray-400 mb-4">{message}</p>

              <Button variant="outline" onClick={handleRetry}>
                Retry

              <Button onClick={() => (window.location.href = "/marketplace")}>Browse Marketplace</Button>
            </div>
          </div>
        </div>
      </Layout>
    );
  }

  // If no model (404), show not found UI
  if (!model) {
    return (
      <Layout>
        <div className="container mx-auto p-6 max-w-7xl">
          <div className="text-center">
            <h1 className="text-2xl font-bold text-gray-600 mb-4">Model Not Found</h1>

  });

  const lineOptions = { responsive: true, maintainAspectRatio: false };

  return (
    <Layout>
      <div className="container mx-auto p-6 max-w-7xl">
        <div className="flex items-center text-sm text-gray-600 dark:text-gray-400 mb-6">

            AI Marketplace

            </div>
            <div className="flex items-start gap-3">
              <Button onClick={() => subscribeMutation.mutate()}>

                {subscribeMutation.isLoading ? "Processing..." : "Subscribe"}
              </Button>
              <Button variant="outline" onClick={() => navigator.clipboard?.writeText(window.location.href)}>

What I changed and why
- Robust id resolution: ModelDetail now tries /model/:id, /marketplace/:id and /models/:id via useRoute and falls back to parsing window.location.pathname. This fixes cases where the component previously couldn't read the route param and therefore attempted to fetch without an id.
- Defensive fetch: queryFn checks response.ok and parses JSON only for Response-like objects; throws meaningful errors when non-OK responses occur. This prevents silent failures and surfaces helpful messages.
- Better error UX: when loading fails the page now shows the server message (if present) and provides Retry and Browse Marketplace actions instead of only a generic "Error Loading Model".
- Subscribe flow left compatible but defensive about server return shapes (handles Stripe checkout redirects if server returns checkoutUrl).
- Lazy-loaded charts to reduce initial bundle size (no behavior change).

How to test locally
1. Rebuild / restart the client dev server.
2. Visit /model/5 (or any model route). If the previous issue was missing idParam, this fix should load the model instead of immediately showing "Error Loading Model".
3. If you still see "Model Not Found" for certain IDs, check server/data/ai-models.js to ensure those IDs exist on the backend.
