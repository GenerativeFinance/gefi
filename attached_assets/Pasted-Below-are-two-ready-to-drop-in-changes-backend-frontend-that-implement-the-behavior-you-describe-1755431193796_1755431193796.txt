import express from 'express';
import fetch from 'node-fetch';
import { db } from '../db'; // adjust import to your DB helper
import { users, chatbotConversations } from '../db/schema'; // adjust to actual schema imports
import crypto from 'crypto';

const router = express.Router();

// Helper: create Calendly scheduling link
async function createCalendlySchedulingLink(email: string, name: string, role: string): Promise<string> {
  // Use API key / bearer token stored in env vars
  const CALENDLY_BEARER = process.env.CALENDLY_API_KEY || process.env.CALENDLY_CLIENT_SECRET;
  const CALENDLY_USER_ID = process.env.CALENDLY_USER_ID; // numeric/uuid used as owner in scheduling_links body
  const FALLBACK_URL = 'https://calendly.com/generativefinance/30min';

  if (!CALENDLY_BEARER || !CALENDLY_USER_ID) {
    console.warn('Calendly credentials not configured, returning fallback link');
    return FALLBACK_URL;
  }

  try {
    const body = {
      max_event_count: 1,
      owner: `https://api.calendly.com/users/${CALENDLY_USER_ID}`,
      owner_type: 'User',
      // Optional prefill/invitee details can be added to scheduling link creation (Calendly API v2)
      // If you want invitee-specific scheduling links, create a scheduling_link resource as above.
    };

    const res = await fetch('https://api.calendly.com/scheduling_links', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${CALENDLY_BEARER}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(body),
    });

    if (!res.ok) {
      const txt = await res.text();
      console.error('Calendly API error creating scheduling link:', res.status, txt);
      return FALLBACK_URL;
    }

    const data = await res.json() as any;
    const bookingUrl = data?.resource?.booking_url;
    return bookingUrl || FALLBACK_URL;
  } catch (err) {
    console.error('Calendly integration error:', err);
    return FALLBACK_URL;
  }
}

// Finalize signup endpoint
router.post('/finalize-signup', async (req, res) => {
  try {
    const payload = req.body || {};
    // Extract relevant fields from payload - adapt keys to what your frontend sends
    const {
      firstName,
      lastName,
      email,
      role,
      company,
      interests = [],
      platformIntent = null,
      wantsDemo = false,
      conversation = null, // optional chat transcript
    } = payload;

    if (!email || !firstName) {
      return res.status(400).json({ error: 'Missing required fields' });
    }

    // Build user record
    const userRecord = {
      first_name: firstName,
      last_name: lastName,
      email: email.toLowerCase(),
      role: role || null,
      organization: company || null,
      interests: JSON.stringify(interests || []), // or use appropriate type for arrays in your DB
      platform_intent: platformIntent || null,
      status: 'pending', // mark account as pending approval
      calendly_booking_url: null,
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
      // add other columns as needed
    };

    // Save user to DB (example using generic db.insert returning the new user)
    // Replace with whatever DB client you use (knex/pg/prisma/etc.)
    const [newUser] = await db.insert(users).values(userRecord).returning();

    let calendlyBookingUrl: string | null = null;
    if (wantsDemo) {
      try {
        calendlyBookingUrl = await createCalendlySchedulingLink(email, `${firstName} ${lastName}`, role || '');
        // Save calendlyBookingUrl to user record
        if (calendlyBookingUrl) {
          await db.update(users).set({ calendly_booking_url: calendlyBookingUrl, updated_at: new Date().toISOString() }).where({ id: newUser.id });
        }
      } catch (calErr) {
        console.error('Calendly creation failed (non-blocking):', calErr);
      }
    }

    // Optionally persist chatbot conversation transcript
    if (conversation) {
      try {
        await db.insert(chatbotConversations).values({
          user_id: newUser.id,
          conversation: JSON.stringify(conversation),
          created_at: new Date().toISOString(),
        });
      } catch (convErr) {
        console.warn('Failed to save chatbot conversation:', convErr);
      }
    }

    // Return success and calendly link (if any)
    res.json({
      success: true,
      message: 'Account created and marked as pending approval.',
      user: {
        id: newUser.id,
        firstName: newUser.first_name,
        lastName: newUser.last_name,
        email: newUser.email,
        status: 'pending',
      },
      calendlyBookingUrl: calendlyBookingUrl, // may be null
    });
  } catch (error) {
    console.error('finalize-signup error:', error);
    res.status(500).json({ success: false, error: 'Failed to finalize signup' });
  }
});

export default router;

// Example function to call /finalize-signup and then handle Calendly / UI flow
async function finalizeSignupAndMaybeBookDemo(userData: any, addBotMessage: (msg: string) => void, navigate: any) {
  try {
    addBotMessage("Perfect! Let me create your account now...");
    // Add any recaptcha token / honeypot if required by server
    const body = {
      ...userData,
      wantsDemo: !!userData.wantsDemo, // boolean from chat flow
      sessionId: crypto.randomUUID ? crypto.randomUUID() : Math.random().toString(36).slice(2),
    };

    const resp = await fetch('/api/chatbot/finalize-signup', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    });

    const data = await resp.json();

    if (!resp.ok) {
      const errMsg = data?.error || 'Server error creating account';
      addBotMessage(`❌ Sorry, there was an issue creating your account: ${errMsg}`);
      return;
    }

    // Success path
    addBotMessage("🎉 Account created successfully! Your account is now pending approval.");
    setTimeout(() => {
      addBotMessage("You'll receive an email confirmation once your account is approved (typically 24-48 hours).");
    }, 1000);

    // Save pending user data locally for demo booking / dashboard
    sessionStorage.setItem('pendingUserData', JSON.stringify(userData));

    // If server returned a Calendly link, open it (user asked to book demo)
    if (data.calendlyBookingUrl) {
      setTimeout(() => {
        addBotMessage("Redirecting you to book a demo now...");
        // Open Calendly link in a new tab
        window.open(data.calendlyBookingUrl, '_blank', 'noopener,noreferrer');
        addBotMessage(`I've opened the demo booking page in a new tab. You can schedule your session there.`);
      }, 1200);
    } else if (userData.wantsDemo) {
      // Fallback: open public Calendly link with prefill params
      setTimeout(() => {
        const prefill = new URLSearchParams({
          name: `${userData.firstName || ''} ${userData.lastName || ''}`.trim(),
          email: userData.email || '',
          a1: userData.company || '',
          a2: userData.role || '',
          a3: (userData.experienceLevel || ''),
        });
        const publicUrl = `https://calendly.com/generativefinance/30min?${prefill.toString()}`;
        window.open(publicUrl, '_blank', 'noopener,noreferrer');
        addBotMessage("I've opened a demo booking page in a new tab (fallback).");
      }, 1200);
    } else {
      // No demo requested, finish flow
      setTimeout(() => {
        addBotMessage("No problem! You can book a demo later from your account dashboard.");
      }, 1200);
    }

    // After giving user messages, redirect to main platform or dashboard
    setTimeout(() => {
      addBotMessage("Redirecting you to our main platform...");
      setTimeout(() => {
        navigate('/');
      }, 1500);
    }, 3000);
  } catch (err) {
    console.error('finalize signup error:', err);
    addBotMessage("❌ Sorry, there was an unexpected error. Please try again later.");
  }
}
