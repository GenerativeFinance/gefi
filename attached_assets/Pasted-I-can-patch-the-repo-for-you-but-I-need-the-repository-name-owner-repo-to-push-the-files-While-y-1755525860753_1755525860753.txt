// Jest + Supertest integration test updated to run migrations before tests.
// It will attempt to run Prisma migrations if USE_PRISMA env var is set,
// otherwise fall back to running sequelize-cli migrations.
//
// Ensure DATABASE_URL is set in CI (GitHub Actions via secrets.DATABASE_URL)

const request = require('supertest');
const { execSync } = require('child_process');
const path = require('path');

let app;
let User;
let sequelize;

beforeAll(async () => {
  // Run migrations appropriate to the stack in this environment.
  try {
    if (process.env.USE_PRISMA === 'true') {
      console.log('Running Prisma migrations (deploy)...');
      execSync('npx prisma migrate deploy', { stdio: 'inherit' });
    } else {
      console.log('Running Sequelize migrations (test env)...');
      // If your project requires a specific env for sequelize-cli, set it above.
      execSync('npx sequelize-cli db:migrate --env test', { stdio: 'inherit' });
    }
  } catch (err) {
    // Fail loudly in CI, but give debugging info locally.
    console.error('Migration step failed:', err);
    throw err;
  }

  // Import app and models after migrations have run
  app = require('../../app'); // adjust if your app entry is elsewhere
  ({ User, sequelize } = require('../../models')); // adjust path as needed
});

afterAll(async () => {
  if (sequelize && typeof sequelize.close === 'function') {
    await sequelize.close();
  }
});

describe('POST /api/signup', () => {
  it('creates a user with Pending status', async () => {
    const payload = {
      email: 'test+pending@example.com',
      password: 'supersecurepassword',
      intent: 'Browse/Learn',
      interests: ['Investing', 'Savings'],
    };

    const res = await request(app).post('/api/signup').send(payload);
    expect(res.statusCode).toBe(201);
    expect(res.body.status).toBe('Pending');

    const dbUser = await User.findOne({ where: { email: payload.email }, raw: true });
    expect(dbUser).not.toBeNull();
    expect(dbUser.status).toBe('Pending');

    // clean up
    await User.destroy({ where: { email: payload.email } });
  });
});
