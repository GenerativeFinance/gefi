Here’s a small, client-only patch to deep-link the “Manage Contract” button on Portfolio AI Models directly to the corresponding contract in Contract Wallet, and auto-open that contract when the wallet page loads. Works out of the box on Replit.

```typescript name=client/src/components/portfolio/StakeholderMetrics.tsx
import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";
import { BarChart3, Link as LinkIcon, Users } from "lucide-react";
import { useLocation } from "wouter";

type Props = {
  model: {
    id: number | string;
    name: string;
    dailyUsers?: number;
    monthlyUsers?: number;
    yearlyUsers?: number;
    stakeholderSharePct?: number; // 0-100
    roiPct?: number; // e.g., 12.5 => +12.5%
    otherStakeholders?: number;
    contractChain?: string; // e.g., "Solana" | "Ethereum" | "Base" | "Polygon"
  };
};

export default function StakeholderMetrics({ model }: Props) {
  const [, setLocation] = useLocation();

  const chainParam =
    (model.contractChain || "").toLowerCase().trim() || "solana";

  const goToContractWallet = () => {
    // Deep-link directly to Contract Wallet with preselection hints
    // Supported params the wallet now reads: contractId, modelId, contractAddress, chain
    const url = `/wallet?contractId=${encodeURIComponent(
      String(model.id)
    )}&model=${encodeURIComponent(model.name)}&chain=${encodeURIComponent(
      chainParam
    )}`;
    setLocation(url);
  };

  return (
    <Card className="border-dashed">
      <CardContent className="p-4">
        <div className="grid grid-cols-1 md:grid-cols-5 gap-4 items-center">
          <div className="space-y-1">
            <p className="text-sm text-muted-foreground">Daily Users</p>
            <p className="text-lg font-semibold">{model.dailyUsers ?? 0}</p>
          </div>
          <div className="space-y-1">
            <p className="text-sm text-muted-foreground">Monthly Users</p>
            <p className="text-lg font-semibold">{model.monthlyUsers ?? 0}</p>
          </div>
          <div className="space-y-1">
            <p className="text-sm text-muted-foreground">Yearly Users</p>
            <p className="text-lg font-semibold">{model.yearlyUsers ?? 0}</p>
          </div>
          <div className="space-y-1">
            <p className="text-sm text-muted-foreground">Stakeholder Share</p>
            <div className="flex items-center gap-2">
              <Badge variant="secondary">
                {(model.stakeholderSharePct ?? 0).toFixed(1)}%
              </Badge>
              <span className="text-xs text-muted-foreground">
                {model.otherStakeholders
                  ? `${model.otherStakeholders} stakeholders`
                  : "No other stakeholders"}
              </span>
            </div>
          </div>
          <div className="space-y-1">
            <p className="text-sm text-muted-foreground">ROI</p>
            <p
              className={`text-lg font-semibold ${
                (model.roiPct ?? 0) >= 0 ? "text-green-600" : "text-red-600"
              }`}
            >
              {(model.roiPct ?? 0) >= 0 ? "+" : ""}
              {(model.roiPct ?? 0).toFixed(1)}%
            </p>
          </div>
        </div>

        <div className="mt-4 flex items-center justify-between">
          <div className="flex items-center gap-2 text-sm text-muted-foreground">
            <Users className="h-4 w-4" />
            <span>{model.contractChain || "Solana"} Contract</span>
          </div>
          <div className="flex gap-2">
            <Button variant="outline" size="sm" className="gap-2">
              <BarChart3 className="h-4 w-4" />
              Analytics
            </Button>
            <Button onClick={goToContractWallet} size="sm" className="gap-2">
              <LinkIcon className="h-4 w-4" />
              Manage Contract
            </Button>
          </div>
        </div>
      </CardContent>
    </Card>
  );
}
```

      // Accept multiple hints

        params.get("contractId") ||
        params.get("modelId") ||
        params.get("id");
      const byAddress = params.get("contractAddress");
      const chain = params.get("chain");

      if (chain) {
        setSelectedChain(String(chain).toLowerCase());
      }

      let target: any | undefined;
      if (byId) {
        target = flContracts.find(
          (c) => String(c.id) === String(byId)
        );
      }
      if (!target && byAddress) {
        const needle = String(byAddress).toLowerCase();
        target = flContracts.find(
          (c) => String(c.contractAddress || "").toLowerCase().includes(needle)
        );
      }

      if (target) {
        // Open the contract dialog by setting selectedContract

Notes
- Manage Contract now deep-links to /wallet with contractId, model name, and chain hint.
- Contract Wallet reads the query string and:
  - Sets the selected chain if provided.
  - Pre-selects and opens the matching contract dialog by id or address.
- No server changes; works in Replit immediately.
