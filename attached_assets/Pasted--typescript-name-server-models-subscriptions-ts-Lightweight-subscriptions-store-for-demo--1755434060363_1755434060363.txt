```typescript name=server/models/subscriptions.ts
/**
 * Lightweight subscriptions store for demo + migration-friendly API.
 *
 * - Default: local JSON store at server/data/subscriptions.json
 * - Optional DB: if you set USE_DATABASE=true and export a `db` client from server/db,
 *   the helper will attempt to use the DB (best-effort - adapt queries to your DB client).
 *
 * This file provides a small abstraction so the rest of the code can call:
 *  - createSubscription(payload)
 *  - findSubscriptionByUserAndModel(userId, modelId)
 *  - listSubscriptionsByUser(userId)
 *  - updateSubscription(id, patch)
 *
 * Replace internals with your preferred DB client (Prisma/Knex/pg) in production.
 */

import fs from "fs/promises";
import path from "path";

const JSON_DB_PATH = path.join(__dirname, "..", "data", "subscriptions.json");

type SubscriptionStatus = "pending" | "active" | "cancelled" | "past_due" | "trialing";

type SubscriptionRecord = {
  id: number;
  userId: string | number;
  modelId: number;
  price?: number | null;
  currency?: string | null;
  status: SubscriptionStatus;
  createdAt: string;
  updatedAt: string;
  metadata?: Record<string, any>;
};

async function ensureJsonStore() {
  try {
    await fs.access(JSON_DB_PATH);
  } catch (e) {
    await fs.mkdir(path.dirname(JSON_DB_PATH), { recursive: true });
    await fs.writeFile(JSON_DB_PATH, JSON.stringify({ lastId: 0, items: [] }, null, 2), "utf8");
  }
}

async function readJsonStore(): Promise<{ lastId: number; items: SubscriptionRecord[] }> {
  await ensureJsonStore();
  const raw = await fs.readFile(JSON_DB_PATH, "utf8");
  return JSON.parse(raw);
}

async function writeJsonStore(store: { lastId: number; items: SubscriptionRecord[] }) {
  await fs.writeFile(JSON_DB_PATH, JSON.stringify(store, null, 2), "utf8");
}

export async function createSubscriptionLocal(payload: {
  userId: string | number;
  modelId: number;
  price?: number | null;
  currency?: string | null;
  status?: SubscriptionStatus;
  metadata?: Record<string, any>;
}): Promise<SubscriptionRecord> {
  const store = await readJsonStore();
  const nextId = store.lastId + 1;
  const now = new Date().toISOString();
  const rec: SubscriptionRecord = {
    id: nextId,
    userId: payload.userId,
    modelId: payload.modelId,
    price: payload.price ?? null,
    currency: payload.currency ?? "USD",
    status: payload.status ?? "active",
    metadata: payload.metadata ?? {},
    createdAt: now,
    updatedAt: now,
  };
  store.lastId = nextId;
  store.items.push(rec);
  await writeJsonStore(store);
  return rec;
}

export async function findSubscriptionLocalByUserAndModel(userId: string | number, modelId: number) {
  const store = await readJsonStore();
  return store.items.find((s) => String(s.userId) === String(userId) && s.modelId === modelId) || null;
}

export async function listSubscriptionsLocalByUser(userId: string | number) {
  const store = await readJsonStore();
  return store.items.filter((s) => String(s.userId) === String(userId));
}

export async function updateSubscriptionLocal(id: number, patch: Partial<SubscriptionRecord>) {
  const store = await readJsonStore();
  const idx = store.items.findIndex((s) => s.id === id);
  if (idx === -1) return null;
  const existing = store.items[idx];
  const updated = { ...existing, ...patch, updatedAt: new Date().toISOString() };
  store.items[idx] = updated;
  await writeJsonStore(store);
  return updated;
}

      const { db } = require("../db");

export async function findSubscriptionByUserAndModel(userId: string | number, modelId: number) {
  if (process.env.USE_DATABASE === "true") {
    try {
      // eslint-disable-next-line @typescript-eslint/no-var-requires

      return rows[0] || null;
    } catch (e) {
      console.warn("DB lookup failed, falling back to local store");
      return findSubscriptionLocalByUserAndModel(userId, modelId);
    }
  } else {
    return findSubscriptionLocalByUserAndModel(userId, modelId);
  }
}

export async function listSubscriptionsByUser(userId: string | number) {
  if (process.env.USE_DATABASE === "true") {
    try {
      // eslint-disable-next-line @typescript-eslint/no-var-requires
      const { db } = require("../db");
      const rows = await db("subscriptions").where({ user_id: String(userId) });
      return rows;
    } catch (e) {
      console.warn("DB list failed, falling back to local store");
      return listSubscriptionsLocalByUser(userId);
    }
  } else {
    return listSubscriptionsLocalByUser(userId);
  }
}

export async function updateSubscription(id: number, patch: Partial<SubscriptionRecord>) {
  if (process.env.USE_DATABASE === "true") {
    try {
      // eslint-disable-next-line @typescript-eslint/no-var-requires
      const { db } = require("../db");
      await db("subscriptions").where({ id }).update(patch);
      const rows = await db("subscriptions").where({ id }).limit(1);
      return rows[0] || null;
    } catch (e) {
      console.warn("DB update failed, falling back to local store");
      return updateSubscriptionLocal(id, patch as any);
    }
  } else {
    return updateSubscriptionLocal(id, patch as any);
  }
}

export default {
  createSubscription,
  findSubscriptionByUserAndModel,
  listSubscriptionsByUser,
  updateSubscription,
};
```

```typescript name=server/utils/payments.ts
/**
 * Lightweight Stripe helper module.
 * - If STRIPE_API_KEY is not set, functions will throw so callers can fallback to demo/local mode.
 *
 * Usage:
 *   const session = await createCheckoutSession({ priceCents, productName, successUrl, cancelUrl, metadata });
 *
 * NOTE: install `stripe` in production to enable this: npm i stripe
 */

type CreateCheckoutParams = {
  priceCents: number;
  currency?: string;
  productName?: string;
  successUrl: string;
  cancelUrl: string;
  metadata?: Record<string, any>;
};

const stripeSecret = process.env.STRIPE_API_KEY;

export async function createCheckoutSession(params: CreateCheckoutParams) {
  if (!stripeSecret) {
    throw new Error("Stripe not configured (STRIPE_API_KEY missing)");
  }
  // dynamic import to avoid failing when stripe is not installed in demo envs
  // eslint-disable-next-line @typescript-eslint/no-var-requires
  const Stripe = require("stripe");
  const stripe = new Stripe(stripeSecret, { apiVersion: "2022-11-15" });

  const session = await stripe.checkout.sessions.create({
    payment_method_types: ["card"],
    mode: "payment",
    line_items: [
      {
        price_data: {
          currency: params.currency || "usd",
          product_data: {
            name: params.productName || "AI Model Subscription",
          },
          unit_amount: params.priceCents,
        },
        quantity: 1,
      },
    ],
    success_url: params.successUrl,
    cancel_url: params.cancelUrl,
    metadata: params.metadata || {},
  });

  return { id: session.id, url: session.url };
}
```

```typescript name=server/routes/subscriptions.ts
/**
 * Register subscription-related routes.
 *
 * Mount this file from your main routes loader (server/routes/gefiRoutes.ts or server/src/index.ts):
 *   const registerSubscriptions = require('./routes/subscriptions').default;
 *   registerSubscriptions(app);
 *
 * Endpoints added:
 *  - POST /api/ai-models/:id/subscribe
 *  - GET  /api/ai-models/:id/subscription
 *  - POST /api/webhooks/stripe  (optional webhook handler for checkout.session.completed)
 *
 * Notes:
 *  - This file uses a small local isAuthenticated guard when no central auth middleware is available.
 *  - It expects aiModelsData to be available at server/data/ai-models.js (the repo already has this file).
 */

import express from "express";
import path from "path";
import bodyParser from "body-parser";

const router = express.Router();

// Local guard (use your app's auth middleware in production)
function ensureAuth(req: any, res: any, next: any) {
  if (req.user || (req as any).user) {
    return next();
  }
  // If your app stores JWT claims differently, adapt this guard to match.
  return res.status(401).json({ message: "Unauthorized" });
}

// dynamic imports so this file remains safe in simple demo runs
// eslint-disable-next-line @typescript-eslint/no-var-requires
const aiModelsData = require(path.join(__dirname, "..", "data", "ai-models.js")).aiModelsData;
// eslint-disable-next-line @typescript-eslint/no-var-requires
const subs = require(path.join(__dirname, "..", "models", "subscriptions")).default;
// eslint-disable-next-line @typescript-eslint/no-var-requires
const payments = require(path.join(__dirname, "..", "utils", "payments"));

/**
 * POST /api/ai-models/:id/subscribe
 * - If Stripe is configured, returns { checkoutUrl } to redirect user to Stripe Checkout.
 * - Otherwise creates a demo active subscription record and returns it.
 */
router.post("/api/ai-models/:id/subscribe", ensureAuth, async (req: any, res: any) => {
  try {
    const modelId = parseInt(req.params.id, 10);
    const userId = req.user?.claims?.sub || req.user?.id || (req as any).user?.id;
    if (!userId) return res.status(401).json({ message: "Unauthorized" });

    const model = aiModelsData.find((m: any) => m.id === modelId);

    // attempt to avoid duplicate active subscription
    const existing = await subs.findSubscriptionByUserAndModel(userId, modelId);
    if (existing && existing.status === "active") {
      return res.json({ success: true, message: "Already subscribed", subscription: existing });
    }

    // If Stripe configured -> create Checkout session and create a pending subscription record
    if (process.env.STRIPE_API_KEY) {
      try {
        const priceCents = Math.round((model?.price ? Number(model.price) : 0) * 100);
        const host = req.headers["x-forwarded-host"] || req.headers.host || "localhost:3000";
        const protocol = req.headers["x-forwarded-proto"] || (req.secure ? "https" : "http");
        const successUrl = `${protocol}://${host}/billing/success?modelId=${modelId}&session_id={CHECKOUT_SESSION_ID}`;
        const cancelUrl = `${protocol}://${host}/billing/cancel`;

        const session = await payments.createCheckoutSession({
          priceCents,
          currency: "usd",
          productName: model?.name || "AI Model Subscription",
          successUrl,
          cancelUrl,
          metadata: { modelId: String(modelId), userId: String(userId) },
        });

        const pending = await subs.createSubscription({
          userId,
          modelId,
          price: model?.price ? Number(model.price) : null,
          currency: "USD",
          status: "pending",
          metadata: { checkoutSessionId: session.id },
        });

        return res.json({ success: true, checkoutUrl: session.url, subscription: pending });
      } catch (stripeErr) {
        console.error("Stripe checkout creation failed:", stripeErr);
        // fallthrough to demo fallback
      }
    }

    // Demo/local fallback: create active subscription immediately
    const created = await subs.createSubscription({
      userId,
      modelId,
      price: model?.price ? Number(model.price) : null,
      currency: "USD",
      status: "active",
      metadata: {},
    });

    return res.json({ success: true, message: "Successfully subscribed (demo mode)", subscription: created });
  } catch (err) {
    console.error("Subscribe endpoint error:", err);
    return res.status(500).json({ success: false, message: "Failed to subscribe to model" });
  }
});

/**
 * GET /api/ai-models/:id/subscription
 * Returns the current user's subscription record for the given model (or null).
 */
router.get("/api/ai-models/:id/subscription", ensureAuth, async (req: any, res: any) => {
  try {
    const modelId = parseInt(req.params.id, 10);
    const userId = req.user?.claims?.sub || req.user?.id || (req as any).user?.id;
    if (!userId) return res.status(401).json({ message: "Unauthorized" });
    const sub = await subs.findSubscriptionByUserAndModel(userId, modelId);
    return res.json({ subscription: sub || null });
  } catch (err) {
    console.error("Get subscription failed:", err);
    return res.status(500).json({ message: "Failed to fetch subscription" });
  }
});

/**
 * Stripe webhook handler (optional)
 * - Posts to /api/webhooks/stripe with Stripe's signature header will be validated.
 * - On checkout.session.completed, the pending subscription is activated.
 *
 * To enable in local dev use stripe CLI to forward webhooks and set STRIPE_WEBHOOK_SECRET.
 */
router.post(
  "/api/webhooks/stripe",
  // receive raw body for signature verification
  express.raw({ type: "application/json" }),
  async (req: any, res: any) => {
    try {
      const stripeSigningSecret = process.env.STRIPE_WEBHOOK_SECRET;
      if (!process.env.STRIPE_API_KEY || !stripeSigningSecret) {
        return res.status(400).send("Stripe not configured");
      }
      // eslint-disable-next-line @typescript-eslint/no-var-requires
      const Stripe = require("stripe");
      const stripe = new Stripe(process.env.STRIPE_API_KEY, { apiVersion: "2022-11-15" });
      const sig = req.headers["stripe-signature"] as string;
      let event;
      try {
        event = stripe.webhooks.constructEvent(req.body, sig, stripeSigningSecret);
      } catch (e: any) {
        console.error("Invalid Stripe webhook signature", e.message);
        return res.status(400).send(`Webhook Error: ${e.message}`);
      }

      if (event.type === "checkout.session.completed") {
        const session = event.data.object;
        const metadata = session.metadata || {};
        const modelId = metadata.modelId ? Number(metadata.modelId) : null;
        const userId = metadata.userId || null;
        const checkoutSessionId = session.id;
        const { findSubscriptionByUserAndModel, updateSubscription } = subs;
        if (userId && modelId) {
          const pending = await findSubscriptionByUserAndModel(userId, modelId);
          if (pending && pending.metadata?.checkoutSessionId === checkoutSessionId) {
            await updateSubscription(pending.id, { status: "active", metadata: { ...pending.metadata, stripeSession: session } });
            console.log("Subscription activated for user", userId, "model", modelId);
          }
        }
      }

      res.json({ received: true });
    } catch (err) {
      console.error("Stripe webhook handling failed", err);
      res.status(500).send("Webhook processing error");
    }
  }
);

export default (app: any) => {
  // mount router into app
  app.use(router);
};
```
