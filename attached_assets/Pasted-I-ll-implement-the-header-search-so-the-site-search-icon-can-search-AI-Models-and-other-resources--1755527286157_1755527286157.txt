```javascript name=src/components/SearchModal.jsx
import React, { useEffect, useState, useRef } from 'react';
import axios from 'axios';
import debounce from 'lodash.debounce';

/*
 Search modal component:
 - Debounces input and queries /api/search?q=...
 - Displays tabs: Models, Strategies, Reports
 - Expect server response shape: { models: [...], strategies: [...], reports: [...] }
 - Each item should include at least: id, name, description, url
*/

export default function SearchModal({ onClose }) {
  const [q, setQ] = useState('');
  const [loading, setLoading] = useState(false);
  const [results, setResults] = useState({ models: [], strategies: [], reports: [] });
  const [activeTab, setActiveTab] = useState('models');
  const mounted = useRef(false);

  // Debounced search function
  const doSearch = useRef(
    debounce(async (term) => {
      if (!term || term.trim().length === 0) {
        setResults({ models: [], strategies: [], reports: [] });
        setLoading(false);
        return;
      }
      setLoading(true);
      try {
        const resp = await axios.get('/api/search', { params: { q: term } });
        // defensive default
        setResults({
          models: resp.data.models || [],
          strategies: resp.data.strategies || [],
          reports: resp.data.reports || [],
        });
      } catch (err) {
        console.error('Search error', err);
      } finally {
        setLoading(false);
      }
    }, 300)
  ).current;

  useEffect(() => {
    mounted.current = true;
    return () => {
      mounted.current = false;
      doSearch.cancel();
    };
  }, [doSearch]);

  useEffect(() => {
    if (!q) {
      setResults({ models: [], strategies: [], reports: [] });
      return;
    }
    doSearch(q);
  }, [q, doSearch]);

  const items = results[activeTab] || [];

  return (
    <div className="search-modal-overlay" role="dialog" aria-modal="true">
      <div className="search-modal">
        <div className="search-header">
          <input
            autoFocus
            value={q}
            onChange={(e) => setQ(e.target.value)}
            placeholder="Search across AI models, trading strategies, reports, and more..."
            className="search-input"
            aria-label="Search"
          />
          <button className="close-btn" onClick={onClose} aria-label="Close search">×</button>
        </div>

        <div className="search-tabs">
          <button className={activeTab === 'models' ? 'active' : ''} onClick={() => setActiveTab('models')}>
            Models ({results.models.length})
          </button>
          <button className={activeTab === 'strategies' ? 'active' : ''} onClick={() => setActiveTab('strategies')}>
            Strategies ({results.strategies.length})
          </button>
          <button className={activeTab === 'reports' ? 'active' : ''} onClick={() => setActiveTab('reports')}>
            Reports ({results.reports.length})
          </button>
        </div>

        <div className="search-body">
          {loading && <div className="search-loading">Searching…</div>}
          {!loading && items.length === 0 && <div className="search-empty">No results</div>}
          <ul className="search-results">
            {items.map((it) => (
              <li key={it.id} className="search-result">
                {/* Open in-app routes if url is internal; otherwise external link */}
                <a
                  href={it.url}
                  onClick={(e) => {
                    // if it's an internal link, prefer SPA navigation (replace with your router link)
                    // Example: if you use react-router-dom, replace <a> with <Link to={it.url}>
                  }}
                >
                  <div className="result-title">{it.name}</div>
                  {it.description && <div className="result-desc">{it.description}</div>}
                  {it.tags && (
                    <div className="result-tags">
                      {it.tags.slice(0, 5).map((t) => (
                        <span key={t} className="tag">{t}</span>
                      ))}
                    </div>
                  )}
                </a>
              </li>
            ))}
          </ul>
        </div>

        <div className="search-footer">
          <div className="popular-searches">
            Popular searches:
            <button onClick={() => setQ('Risk Assessment')}>Risk Assessment</button>
            <button onClick={() => setQ('Trading Bots')}>Trading Bots</button>
            <button onClick={() => setQ('Portfolio')}>Portfolio</button>
          </div>
        </div>
      </div>
    </div>
  );
}
```

```javascript name=server/routes/search.prisma.js
// Express route using Prisma client. Place in your routes and mount similarly.
// Assumes you have `prisma` client exported from a central file (e.g., server/prisma.js)

const express = require('express');
const router = express.Router();
const prisma = require('../prisma'); // adjust path to your prisma client instance

router.get('/search', async (req, res) => {
  const q = (req.query.q || '').trim();
  if (!q) {
    return res.json({ models: [], strategies: [], reports: [] });
  }

  try {
    // Simple case-insensitive contains search
    const modelsRaw = await prisma.aiModel.findMany({
      where: {
        OR: [
          { name: { contains: q, mode: 'insensitive' } },
          { description: { contains: q, mode: 'insensitive' } },
        ],
      },
      orderBy: { updatedAt: 'desc' },
      take: 20,
      select: { id: true, name: true, description: true, slug: true, tags: true },
    });

    const models = modelsRaw.map((m) => ({
      id: m.id,
      name: m.name,
      description: m.description,
      url: `/models/${m.slug || m.id}`,
      tags: m.tags || [],
    }));

    const strategies = [];
    const reports = [];

    res.json({ models, strategies, reports });
  } catch (err) {
    console.error('Prisma search error', err);
    res.status(500).json({ error: 'Search failed' });
  }
});

module.exports = router;
```
