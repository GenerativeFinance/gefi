Example implementation (MeiliSearch) — server side
- An Express route + search service example you can drop in.

server/services/search.ts
```typescript
import { MeiliSearch } from "meilisearch";

const MEILI_HOST = process.env.MEILI_HOST || "http://127.0.0.1:7700";
const MEILI_KEY = process.env.MEILI_MASTER_KEY || "";

const client = new MeiliSearch({ host: MEILI_HOST, apiKey: MEILI_KEY });

export async function indexDocument(indexName: string, doc: any) {
  const index = client.index(indexName);
  return index.addDocuments([doc]);
}

export async function updateDocument(indexName: string, doc: any) {
  const index = client.index(indexName);
  return index.addDocuments([doc]); // addDocuments upserts
}

export async function deleteDocument(indexName: string, id: string) {
  const index = client.index(indexName);
  return index.deleteDocument(id);
}

export async function searchIndex(indexName: string, q: string, opts = {}) {
  const index = client.index(indexName);
  return index.search(q, opts);
}
```

server/routes/search.ts (single endpoint, grouped results)
```typescript
import express from "express";
import { searchIndex } from "../services/search";

const router = express.Router();

/**
 * GET /api/search?q=...&types=models,strategies&limit=10
 * types: comma-separated list of indexes to query
 */
router.get("/", async (req, res) => {
  try {
    const q = String(req.query.q || "");
    const types = String(req.query.types || "models,strategies,reports,users,funding,bounties")
      .split(",")
      .map((t) => t.trim())
      .filter(Boolean);
    const limit = parseInt(String(req.query.limit || "6"), 10);

    if (!q || q.trim().length === 0) {
      return res.json({ q, results: {} });
    }

    // Parallel searches to keep latency low
    const promises = types.map(async (type) => {
      const r = await searchIndex(type, q, { limit });
      return { type, hits: r.hits, nbHits: r.nbHits, processingTimeMs: r.processingTimeMs };
    });

    const results = await Promise.all(promises);
    const grouped: Record<string, any> = {};
    results.forEach((r) => (grouped[r.type] = r));

    res.json({ q, results: grouped });
  } catch (err) {
    console.error("GET /api/search error:", err);
    res.status(500).json({ error: String(err) });
  }
});

export default router;
```

If you prefer Postgres full-text search (no extra service)
- Add a tsvector column on the table(s), update it on insert/update, and create GIN index.
- Example migration for Models (Postgres / SQL):
```sql
ALTER TABLE models ADD COLUMN document_tsv tsvector;

UPDATE models SET document_tsv = to_tsvector('english', coalesce(title,'') || ' ' || coalesce(description,''));

CREATE INDEX models_document_tsv_idx ON models USING GIN(document_tsv);

-- trigger to update tsvector on change
CREATE FUNCTION models_tsvector_trigger() RETURNS trigger AS $$
begin
  new.document_tsv := to_tsvector('english', coalesce(new.title,'') || ' ' || coalesce(new.description,''));
  return new;
end
$$ LANGUAGE plpgsql;

CREATE TRIGGER tsvectorupdate BEFORE INSERT OR UPDATE
  ON models FOR EACH ROW EXECUTE PROCEDURE models_tsvector_trigger();
```
Then search with:
```sql
SELECT id, title, ts_rank(document_tsv, plainto_tsquery('english', $1)) AS rank
FROM models
WHERE document_tsv @@ plainto_tsquery('english', $1)
ORDER BY rank DESC
LIMIT 20;
```
