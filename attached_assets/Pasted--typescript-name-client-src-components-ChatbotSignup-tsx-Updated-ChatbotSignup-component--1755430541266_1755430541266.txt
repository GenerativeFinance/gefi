  lastName?: string;
  role?: string;
  company?: string;
  experienceLevel?: string;
  wantsDemo?: boolean;
  [key: string]: any;

  sender: "bot" | "user";
  message: string | React.ReactNode;
  timestamp: Date;
  isTyping?: boolean;

export default function ChatbotSignup({ onComplete, onBack }: { onComplete?: (userData: UserData) => void; onBack?: () => void }) {
  const [, navigate] = useNavigate ? useNavigate() : [null, () => {}];
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [currentStep, setCurrentStep] = useState(0);
  const [isCreatingAccount, setIsCreatingAccount] = useState(false);
  const [recaptchaToken, setRecaptchaToken] = useState<string | null>(null);
  // Example userData state - in real file this will be built up across steps
  const [userData, setUserData] = useState<UserData>({ email: "", firstName: "", lastName: "" });

  const addBotMessage = (message: string | React.ReactNode, isTyping = false) => {
    const newMessage: ChatMessage = {
      id: `bot-${Date.now()}`,
      sender: "bot",
      message,
      timestamp: new Date(),
      isTyping,
    };
    setMessages((prev) => [...prev, newMessage]);
  };

  const addUserMessage = (message: string) => {
    const newMessage: ChatMessage = {
      id: `user-${Date.now()}`,
      sender: "user",
      message,
      timestamp: new Date(),
    };
    setMessages((prev) => [...prev, newMessage]);
  };

  /**
   * createPendingAccount
   *
   * Centralized account creation logic. Posts to /api/chatbot/signup/complete (canonical).
   * On success:
   *  - stores pending user info in sessionStorage
   *  - adds chat messages
   *  - redirects to /account-pending
   */
  const createPendingAccount = async (data: UserData) => {
    setIsCreatingAccount(true);
    try {
      const sessionId =
        typeof window !== "undefined"
          ? sessionStorage.getItem("chatbot_session_id") ||
            (typeof crypto !== "undefined" && (crypto as any).randomUUID ? (crypto as any).randomUUID() : `${Date.now()}`)
          : null;

      const payload = {
        ...data,
        wantsDemo: data.wantsDemo ?? false,
        sessionId,
        recaptchaToken: recaptchaToken || "",
        honeypot: "",
      };

      // Primary endpoint
      const target = "/api/chatbot/signup/complete";

      const res = await fetch(target, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify(payload),
      });

      if (res.ok) {
        const result = await res.json();

        const pendingUserData = {
          id: result?.id || data.email || null,
          email: data.email,
          firstName: data.firstName,
          lastName: data.lastName,
          role: data.role || "user",
          ...result,
        };

        try {
          sessionStorage.setItem("pendingUserData", JSON.stringify(pendingUserData));
        } catch (e) {
          console.warn("Could not persist pending user data to sessionStorage", e);
        }

        addBotMessage("🎉 Account created successfully! Your account is now pending approval.");
        addBotMessage("You'll receive an email once your account is approved. In the meantime you can schedule a demo.");

        setTimeout(() => {
          navigate("/account-pending");
        }, 900);
      } else {
        let errorMessage = `Server error: ${res.status} ${res.statusText}`;
        try {
          const parsed = await res.json();
          errorMessage = parsed.message || parsed.error || errorMessage;
        } catch {
          // ignore parse error
        }
        addBotMessage(`❌ Sorry, there was an error creating your account: ${errorMessage}`);
      }
    } catch (err) {
      console.error("Signup error:", err);
      addBotMessage("❌ Sorry, there was an error creating your account. Please try again.");
    } finally {
      setIsCreatingAccount(false);
    }

      if (response.ok) {
        addBotMessage("✅ Email verified successfully! Now creating your account...");
        await createPendingAccount(userData);
      } else {
        let err = "Verification failed";
        try {
          const d = await response.json();
          err = d.message || err;
        } catch {}
        addBotMessage(`❌ ${err}`);
      }
    } catch (err) {

import type { Express, Request, Response } from "express";

/**
 * Lightweight compatibility route:
 * Accepts POST /api/auth/complete-chatbot-signup and forwards the request
 * to the canonical /api/chatbot/signup/complete endpoint.
 *
 * This helps older clients that still post to /api/auth/complete-chatbot-signup
 * while the server expects /api/chatbot/signup/complete.
 *
 * Note: This uses global fetch (Node 18+). If your runtime lacks global fetch,
 * install node-fetch and replace the call accordingly.
 */
export default function registerAuthCompat(app: Express) {
  app.post("/api/auth/complete-chatbot-signup", async (req: Request, res: Response) => {
    try {
      // Determine protocol and host to forward to same server
      const proto = (req as any).protocol || "http";
      const host = req.get("host") || "localhost:3000";
      const targetUrl = `${proto}://${host}/api/chatbot/signup/complete`;

      const forwardHeaders: Record<string, string> = {
        "Content-Type": req.get("content-type") || "application/json",
      };
      if (req.get("authorization")) {
        forwardHeaders["Authorization"] = req.get("authorization") as string;
      }

      const fetchResponse = await fetch(targetUrl, {
        method: "POST",
        headers: forwardHeaders,
        body: JSON.stringify(req.body),
      });

      const contentType = fetchResponse.headers.get("content-type") || "application/json";
      res.status(fetchResponse.status).set("Content-Type", contentType);
      const text = await fetchResponse.text();

      if (contentType.includes("application/json")) {
        try {
          return res.send(JSON.parse(text));
        } catch {
          return res.send(text);
        }
      }

      return res.send(text);
    } catch (err) {
      console.error("Error forwarding /api/auth/complete-chatbot-signup:", err);
      return res.status(500).json({ message: "Failed to forward signup request" });
    }
  });
}
