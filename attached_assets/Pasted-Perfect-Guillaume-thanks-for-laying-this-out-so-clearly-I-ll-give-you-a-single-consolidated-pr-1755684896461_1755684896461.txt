### App Marketplace Page (`AppMarketplace.jsx`)

```jsx
import { useState } from "react";

const mockApps = [
  { id: 1, name: "Slack", category: "Messaging", description: "Team communication platform", icon: "/slack.png" },
  { id: 2, name: "Asana", category: "Project Management", description: "Task & project tracking", icon: "/asana.png" },
  { id: 3, name: "AWS", category: "Cloud Services", description: "Amazon Web Services cloud tools", icon: "/aws.png" },
  { id: 4, name: "TensorFlow", category: "AI Tools", description: "Deep learning framework", icon: "/tf.png" }
];

export default function AppMarketplace() {
  const [installed, setInstalled] = useState([]);

  const handleInstall = (app) => {
    if (!installed.includes(app.id)) {
      setInstalled([...installed, app.id]);
    }
  };

  return (
    <div className="p-6 text-white bg-gray-950 min-h-screen">
      <h1 className="text-2xl font-bold mb-4">App Marketplace</h1>
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        {mockApps.map(app => (
          <div key={app.id} className="bg-gray-800 p-4 rounded-xl shadow">
            <img src={app.icon} alt={app.name} className="h-10 mb-2" />
            <h2 className="text-lg font-semibold">{app.name}</h2>
            <p className="text-sm text-gray-400">{app.description}</p>
            <button 
              onClick={() => handleInstall(app)} 
              className="mt-3 bg-purple-600 px-3 py-1 rounded hover:bg-purple-700"
            >
              {installed.includes(app.id) ? "Installed" : "Install"}
            </button>
          </div>
        ))}
      </div>
    </div>
  );
}
```

### Integrations Page (`Integrations.jsx`)

```jsx
import { useState } from "react";

const mockInstalled = [
  { id: 1, name: "Slack", status: "Connected" },
  { id: 3, name: "AWS", status: "Error" }
];

export default function Integrations() {
  const [apps, setApps] = useState(mockInstalled);

  const handleToggle = (id) => {
    setApps(apps.map(a => 
      a.id === id ? { ...a, status: a.status === "Connected" ? "Disabled" : "Connected" } : a
    ));
  };

  return (
    <div className="p-6 text-white bg-gray-950 min-h-screen">
      <h1 className="text-2xl font-bold mb-4">Integrations</h1>
      <div className="flex flex-col gap-3">
        {apps.map(app => (
          <div key={app.id} className="flex justify-between items-center bg-gray-800 p-4 rounded-xl shadow">
            <div>
              <h2 className="font-semibold">{app.name}</h2>
              <p className="text-xs text-gray-400">Status: {app.status}</p>
            </div>
            <div className="flex gap-2">
              <button onClick={() => handleToggle(app.id)} className="bg-purple-600 px-3 py-1 rounded hover:bg-purple-700">
                {app.status === "Connected" ? "Disable" : "Enable"}
              </button>
              <button className="bg-red-600 px-3 py-1 rounded hover:bg-red-700">Uninstall</button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```
