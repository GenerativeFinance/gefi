```javascript name=routes/signup.js
// Express route for signup. Adjust path/exports to fit your router setup.
const express = require('express');
const router = express.Router();
const { User } = require('../models'); // adjust path to your models index
const bcrypt = require('bcrypt');
const { body, validationResult } = require('express-validator');

router.post('/signup',
  // Basic validation
  body('email').isEmail(),
  body('password').isLength({ min: 8 }),
  async (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }

    const { email, password, intent, interests } = req.body;

    try {
      // Prevent duplicate accounts
      const existing = await User.findOne({ where: { email } });
      if (existing) {
        return res.status(409).json({ error: 'Email already registered' });
      }

      const passwordHash = await bcrypt.hash(password, 10);

      // Create the user with status 'Pending' (model default will do this)
      const newUser = await User.create({
        email,
        passwordHash,
        // store metadata (optional)
        intent: intent || null,
        interests: interests ? JSON.stringify(interests) : null,
        status: 'Pending', // explicit for clarity
      });

      // Optionally generate and send verification token here
      // const token = await createEmailVerificationToken(newUser.id);
      // await sendVerificationEmail(email, token);

      // Return a safe public representation (do not include passwordHash)
      return res.status(201).json({
        id: newUser.id,
        email: newUser.email,
        status: newUser.status,
        message: 'Account created (Pending). Please check your email to verify.',
      });
    } catch (err) {
      console.error('Signup error', err);
      return res.status(500).json({ error: 'Server error' });
    }
  }
);

module.exports = router;
```

````markdown name=src/components/ChatbotSignup.js
// Frontend chatbot signup handler (React example). Integrate into your chatbot flow.
// This component or handler triggers when the chatbot says "Perfect! Let me create your account now..."
import React, { useState } from 'react';
import axios from 'axios';

export default function ChatbotSignup({ collectedAnswers, onComplete }) {
  // collectedAnswers example: { email, password, interests, intent, captchaAnswer }
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const createPendingAccount = async () => {
    setLoading(true);
    setError(null);
    try {
      const payload = {
        email: collectedAnswers.email,
        password: collectedAnswers.password,
        intent: collectedAnswers.intent,
        interests: collectedAnswers.interests,
      };

      const resp = await axios.post('/api/signup', payload); // match backend route
      // resp.data.status should be 'Pending'
      const { status } = resp.data;
      if (status === 'Pending') {
        // Let the chatbot show a success message and continue flow
        onComplete({
          success: true,
          message: 'Account created. Please check your email to verify and activate your account.',
          account: { id: resp.data.id, email: resp.data.email, status },
        });
      } else {
        onComplete({
          success: true,
          message: `Account created with status: ${status}`,
          account: resp.data,
        });
      }
    } catch (err) {
      console.error(err);
      setError(err.response?.data?.error || 'Failed to create account');
      onComplete({ success: false, message: 'Failed to create account' });
    } finally {
      setLoading(false);
    }
  };

  // Hook into the point where chatbot says "Perfect! Let me create your account now..."
  // For example call createPendingAccount() when that message is emitted/acknowledged.
  return (
    <div>
      <button onClick={createPendingAccount} disabled={loading}>
        {loading ? 'Creating account...' : 'Create my account'}
      </button>
      {error && <div className="error">{error}</div>}
    </div>
  );
}
````
