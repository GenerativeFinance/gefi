report_types = {

layouts = {
    "1": "Portrait",
    "2": "Landscape"
}

# Prompt user for report selection
print("Select a report type:")
for key, value in report_types.items():
    print(f"{key}. {value}")
report_choice = input("Enter the number of your choice: ")

selected_report = report_types.get(report_choice)
if not selected_report:
    print("Invalid choice.")
else:
    # Check if report is ready
    status = check_report_status(selected_report)
    if status["status"] == "Ready":
        # Prompt for visualization
        print("\nSelect visualization type:")
        for key, value in visualizations.items():
            print(f"{key}. {value}")
        viz_choice = input("Enter the number of your choice: ")

        # Prompt for layout
        print("\nSelect PDF layout:")
        for key, value in layouts.items():
            print(f"{key}. {value}")
        layout_choice = input("Enter the number of your choice: ")

        # Download the report
        selected_viz = visualizations.get(viz_choice, "Table")
        selected_layout = layouts.get(layout_choice, "Portrait")
        download_report(selected_report, selected_viz, selected_layout)
    else:
        print("Report is not ready yet.")
