export default function ModelDetail() {
  // Try known route patterns (made the param name generic so it can be slug or id)
  const [, modelRouteParams] = useRoute("/model/:idOrSlug");
  const [, marketplaceRouteParams] = useRoute("/marketplace/:id");
  const [, aiModelsRouteParams] = useRoute("/ai-models/:id");

  // Resolve idParam from route params or fallback to pathname parsing
  const idParamRaw =
    modelRouteParams?.idOrSlug ??
    marketplaceRouteParams?.id ??
    aiModelsRouteParams?.id ??
    (typeof window !== "undefined"
      ? (window.location.pathname.match(/\/(?:model|marketplace|models|ai-models)\/([^/]+)/) || [])[1]
      : undefined);

  const idParam = idParamRaw ? String(idParamRaw) : undefined;

  /* ...existing hooks: queryClient, toast, data fetching helpers ... */

  // Fetch model: attempt numeric id first, otherwise try fetching by slug
  useEffect(() => {
    if (!idParam) return;

    const fetchModel = async () => {
      try {
        // if idParam looks like a number, attempt numeric fetch
        if (!Number.isNaN(Number(idParam))) {
          await refetch({ queryKey: ["api", "ai-models", idParam] });
        } else {
          // non-numeric -> treat as slug; call slug endpoint
          // This endpoint is added in backend patch below: GET /api/ai-models/slug/:slug
          const res = await fetch(`/api/ai-models/slug/${encodeURIComponent(idParam)}`);
          if (!res.ok) throw new Error("Model not found");
          const json = await res.json();
          // mutate the query cache / set local state accordingly (assumes existing query keys)
          queryClient.setQueryData(["api", "ai-models", String(json.id)], json);
          // set model into local query cache used by the page (if you use react-query useQuery for model, adjust accordingly)
          // you may also set a direct component state if needed:
          // setModel(json);
        }
      } catch (e) {
        // allow existing error handling to handle missing models
        console.error("Failed to load model by slug:", e);
      }
    };

    fetchModel();
  }, [idParam, queryClient, refetch]);

  // If we have a loaded model, prefer the canonical path (redirect to canonicalPath or slug-based canonical URL)
  useEffect(() => {

      const cur = window.location.pathname;
      // Avoid redirecting if we're already on the canonical path
      if (cur !== canonical) {
        window.location.href = canonical;
        return;
      }
    }
  }, [model]);

/**
 * GET /api/ai-models/slug/:slug
 * Return the model with the given slug (or 404).
 */
router.get("/slug/:slug", async (req, res) => {
  try {
    const { slug } = req.params;
    if (!slug) return res.status(400).json({ error: "Missing slug" });

    const model = await getModelBySlug(String(slug));
    if (!model) return res.status(404).json({ error: "Model not found" });

    // Ensure canonicalPath is present in the response where possible
    // e.g. model.canonicalPath = model.canonicalPath || `/model/${model.slug || model.id}`;
    res.json(model);
  } catch (err) {
    console.error("Error fetching model by slug:", err);
    res.status(500).json({ error: "Internal server error" });
  }
});

export default router;

Notes / migration steps
- Frontend:
  - marketplace.tsx: handleModelClick now computes a preferred path: model.canonicalPath (if present), else /model/{slug} (if slug present), else /model/{id}.
  - model-detail.tsx: route param is now generic (idOrSlug). The page attempts numeric id fetch first; if the route param is not numeric it calls the new backend endpoint GET /api/ai-models/slug/:slug. After model loads, the page will redirect the browser to the canonical path if the model exposes canonicalPath or a slug-based canonical URL. This ensures the canonical URL is used site-wide.
- Backend:
  - Added an example route GET /api/ai-models/slug/:slug which should look up a model by slug. Hook it into your real controller/DB layer; the example references getModelBySlug in controllers/aiModelsController (you may already have such a function — if not add one).
