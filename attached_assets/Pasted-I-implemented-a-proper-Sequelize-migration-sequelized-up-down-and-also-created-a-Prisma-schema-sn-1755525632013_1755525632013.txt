```javascript name=migrations/20250818-add-status-to-users.js
 *
 * This uses a Postgres ENUM type. The down migration removes the column and drops the enum type.
 *
 * Usage (sequelize-cli):
 *   npx sequelize-cli db:migrate --env <your-env>
 */

  up: async (queryInterface, Sequelize) => {
    // Add enum column (creates underlying Postgres enum type named: "enum_users_status")
    await queryInterface.addColumn('users', 'status', {
      type: Sequelize.ENUM('Pending', 'Active', 'Suspended', 'Deleted'),
      allowNull: false,
      defaultValue: 'Pending',
    });

    // Optional: if you want a DB-level check constraint instead of enum, you'd run raw SQL here.
    // For Postgres the ENUM above is usually sufficient.
  },

  down: async (queryInterface, Sequelize) => {
    // Remove column
    await queryInterface.removeColumn('users', 'status');

    // Drop enum type that Sequelize created (name depends on table/column)
    // The default naming convention is: enum_<table>_<column>
```

```javascript name=routes/signup.js
// Express route for signup (Sequelize). Merge/replace with your existing handler.
// Make sure to adjust model import path to your project structure.

const express = require('express');
const router = express.Router();
const bcrypt = require('bcrypt');
const { body, validationResult } = require('express-validator');
const { User } = require('../models'); // adjust path to your models index

router.post(
  '/api/signup',
  body('email').isEmail(),
  body('password').isLength({ min: 8 }),
  async (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }

    const { email, password, intent, interests } = req.body;

    try {
      // Prevent duplicates
      const existing = await User.findOne({ where: { email } });
      if (existing) {
        return res.status(409).json({ error: 'Email already registered' });
      }

      const passwordHash = await bcrypt.hash(password, 10);

      // Create user with explicit status 'Pending' (model default will also handle this)
      const newUser = await User.create({
        email,
        passwordHash,
        intent: intent || null,
        interests: interests || null,
        status: 'Pending',
      });

      // TODO: create & send verification token/email here; see note below.

      return res.status(201).json({
        id: newUser.id,
        email: newUser.email,
        status: newUser.status,
        message: 'Account created (Pending). Please check your email to verify.',
      });
    } catch (err) {
      console.error('Signup error:', err);
      return res.status(500).json({ error: 'Server error' });
    }
  }
);

module.exports = router;
```

```javascript name=src/components/ChatbotSignup.js
// Minimal frontend React handler to call the new signup endpoint.
// Call createPendingAccount() when the chatbot reaches the "create account" step.

import React, { useState } from 'react';
import axios from 'axios';

export default function ChatbotSignup({ collectedAnswers, onComplete }) {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const createPendingAccount = async () => {
    setLoading(true);
    setError(null);

    try {
      const payload = {
        email: collectedAnswers.email,
        password: collectedAnswers.password,
        intent: collectedAnswers.intent,
        interests: collectedAnswers.interests,
      };

      const resp = await axios.post('/api/signup', payload);
      const { status } = resp.data;

      if (status === 'Pending') {
        onComplete({
          success: true,
          message:
            'Perfect! Your account was created in Pending status. Please check your email for a verification link to activate it.',
          account: { id: resp.data.id, email: resp.data.email, status },
        });
      } else {
        onComplete({ success: true, message: `Account created with status ${status}` });
      }
    } catch (err) {
      console.error(err);
      const errMsg = err.response?.data?.error || 'Failed to create account';
      setError(errMsg);
      onComplete({ success: false, message: errMsg });
    } finally {
      setLoading(false);
    }
  };

  return (
    <div>
      <button onClick={createPendingAccount} disabled={loading}>
        {loading ? 'Creating account...' : 'Create my account'}
      </button>
      {error && <div className="error">{error}</div>}
    </div>
  );
}
```
