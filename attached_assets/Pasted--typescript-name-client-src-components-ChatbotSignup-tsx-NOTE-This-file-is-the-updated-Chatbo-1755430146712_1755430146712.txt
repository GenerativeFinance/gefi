      const payload = {

        wantsDemo: userData.wantsDemo ?? false,
        sessionId: (window && (sessionStorage.getItem('chatbot_session_id') || (crypto && crypto.randomUUID && crypto.randomUUID()))) || null,

      // The server registers /api/chatbot/signup/complete (see server/routes/chatbotRoutes.ts)
      const signupResponse = await fetch('/api/chatbot/signup/complete', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(payload),
      });

      if (signupResponse.ok) {
        const result = await signupResponse.json();

        // Persist minimal user info so AccountPending page can read it
        const pendingUserData = {
          id: result?.id || userData.email || null,
          email: userData.email,
          firstName: userData.firstName,
          lastName: userData.lastName,
          role: userData.role || 'user',
          // any other fields server returned can be stored too
          ...result,
        };
        try {
          sessionStorage.setItem('pendingUserData', JSON.stringify(pendingUserData));
        } catch (e) {
          console.warn('Could not persist pending user data to sessionStorage', e);
        }

        // Inform the user in chat
        addBotMessage("🎉 Account created successfully! Your account is now pending approval.");
        addBotMessage("You'll receive an email once your account is approved. In the meantime you can schedule a demo.");

        // Redirect to account pending page so user sees next steps and Calendly invite

          navigate('/account-pending');

      } else {

        try {
          const errorData = await signupResponse.json();
          errorMessage = errorData.message || errorData.error || errorMessage;
        } catch (e) {
          // ignore parse error
        }
        addBotMessage(`❌ Sorry, there was an error creating your account: ${errorMessage}`);
      }
    } catch (error) {
      console.error('Signup error:', error);
      addBotMessage("❌ Sorry, there was an error creating your account. Please try again.");
    } finally {
      setIsCreatingAccount(false);
    }

      }
    } catch (err) {
      console.error('verify error', err);
      addBotMessage("❌ There was an error verifying your code. Please try again.");
    }
  };

  // 2) If security step passed and component wants to create account immediately
  const handleSecurityPassedCreateAccount = async (userData: UserData) => {
    addBotMessage("Perfect! Let me create your account now...");
    await createPendingAccount(userData);
  };

  // Demo booking handlers still open Calendly in a new tab — account creation and redirect to account-pending are handled above.

    } finally {

  };

  // ... rest of component rendering and handlers remain unchanged, but should call the functions above

  return (
