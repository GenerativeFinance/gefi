+import path from 'path';
+import fs from 'fs/promises';
+import puppeteer from 'puppeteer';
+import Mustache from 'mustache';
+import { uploadToS3 } from '../utils/storage';
+import { updateReportStatus, getReportById } from '../db/reports'; // placeholder helpers - adapt to your DB helper
+import { db, reports as reportsTable } from '../db'; // adjust according to your project's DB exports
+
+export interface ReportJobPayload {
+  reportId: string | number;
+  templateId: string;
+  data: any;
+  options?: {
+    pageSize?: 'A4' | 'Letter';
+    margin?: { top: string; right: string; bottom: string; left: string };
+  };
+}
+
+/**
+ * Process a report job:
+ * - Render the HTML template with provided data
+ * - Launch Puppeteer and render PDF
+ * - Upload to S3 (or local stub) and update the DB record
+ */
+export async function processReportJob(job: ReportJobPayload) {
+  const { reportId, templateId, data, options = {} } = job;
+  try {
+    // Mark report as rendering
+    if (typeof updateReportStatus === 'function') {
+      await updateReportStatus(reportId, { status: 'rendering', updated_at: new Date().toISOString() });
+    } else {
+      // Best-effort DB update: try to use generic db object if available
+      try {
+        await db.update(reportsTable).set({ status: 'rendering', updated_at: new Date().toISOString() }).where({ id: reportId });
+      } catch (err) {
+        // ignore if DB update not available
+        console.warn('DB update (rendering) failed:', (err as Error).message);
+      }
+    }
+
+    // Load template file
+    const templatePath = path.join(__dirname, '..', 'templates', `${templateId}.html`);
+    const templateHtml = await fs.readFile(templatePath, 'utf8');
+
+    // Render HTML using Mustache (data should include base64 chart images and HTML table)
+    const renderedHtml = Mustache.render(templateHtml, data);
+
+    // Write to a temp HTML file
+    const tmpHtmlPath = path.join('/tmp', `report-${reportId}.html`);
+    await fs.writeFile(tmpHtmlPath, renderedHtml, 'utf8');
+
+    // Launch Puppeteer
+    const browser = await puppeteer.launch({
+      args: ['--no-sandbox', '--disable-setuid-sandbox'],
+    });
+    const page = await browser.newPage();
+
+    // Enable larger timeout and wait for network idle so images load
+    await page.goto(`file://${tmpHtmlPath}`, { waitUntil: 'networkidle0', timeout: 60000 });
+
+    const pdfBuffer = await page.pdf({
+      format: options.pageSize || 'A4',
+      printBackground: true,
+      margin: options.margin || { top: '20mm', right: '12mm', bottom: '20mm', left: '12mm' },
+    });
+
+    await browser.close();
+
+    // Upload to S3 (or write to disk if S3 not configured)
+    const s3Key = `reports/${reportId}.pdf`;
+    const pdfUrl = await uploadToS3(pdfBuffer, s3Key, 'application/pdf');
+
+    // Update DB record to completed
+    if (typeof updateReportStatus === 'function') {
+      await updateReportStatus(reportId, { status: 'completed', pdf_url: pdfUrl, completed_at: new Date().toISOString() });
+    } else {
+      try {
+        await db.update(reportsTable).set({ status: 'completed', pdf_url: pdfUrl, updated_at: new Date().toISOString() }).where({ id: reportId });
+      } catch (err) {
+        console.warn('DB update (completed) failed:', (err as Error).message);
+      }
+    }
+
+    return { success: true, pdfUrl };
+  } catch (err) {
+    console.error('Report generation failed:', (err as Error).message);
+    try {
+      if (typeof updateReportStatus === 'function') {
+        await updateReportStatus(reportId, { status: 'failed', error: String(err) });
+      } else {
+        await db.update(reportsTable).set({ status: 'failed', error: String(err), updated_at: new Date().toISOString() }).where({ id: reportId });
+      }
+    } catch (e) {
+      // ignore
+    }
+    return { success: false, error: String(err) };
+  }
+}
