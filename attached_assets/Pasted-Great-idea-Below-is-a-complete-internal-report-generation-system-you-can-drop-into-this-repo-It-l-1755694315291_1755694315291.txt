Great idea. Below is a complete, internal report-generation system you can drop into this repo. It lets any report page enqueue or synchronously generate a PDF from a template, stores the file under storage/reports, exposes it for download, and lists it in the Storage Management UI via a new storage API.

What you get
- Server APIs
  - POST /api/reports/generate: build HTML from template, render to PDF, save under storage/reports, persist metadata, return download link.
  - GET /api/reports: list previously generated reports with metadata.
  - GET /api/reports/templates: list available templates and expected input fields.
  - GET /api/reports/:id/status: returns status and metadata.
  - GET /api/reports/:id/download: stream the PDF.
  - GET /api/storage/files: lists all files under storage, including generated PDFs, so they show in your Storage UI.
- Templates included
  - performance, risk, compliance (mapped to the pages in your screenshots).
- Client helpers
  - generateReport() and poll helpers for on-click “Export” buttons in any report page.
  - Reusable <GenerateReportButton />.

```typescript name=server/templates/reportTemplates.ts
import { escape } from "html-escaper";

export type ReportTemplateKey = "performance" | "risk" | "compliance";

export type ReportInput = {
  title?: string;
  period?: string;
  generatedBy?: string;
  generatedAt?: string; // ISO
  summary?: string;
  metrics?: Array<{ label: string; value: string }>;
  sections?: Array<{ heading: string; body: string }>;
};

function baseHTML(title: string, body: string) {
  // Simple, print-friendly HTML with dark-friendly colors but white page background for PDF
  return `<!doctype html>
<html>
<head>
  <meta charset="utf-8" />
  <title>${escape(title)}</title>
  <meta name="viewport" content="width=device-width, initial-scale=1" />
  <style>
    :root {
      --bg: #ffffff;
      --fg: #0f172a; /* slate-900 */
      --muted: #475569; /* slate-600 */
      --accent: #6d28d9; /* violet-700 */
      --good: #16a34a;
      --warn: #f59e0b;
      --bad: #dc2626;
      --card: #f8fafc;
      --border: #e2e8f0;
    }
    @page { size: A4; margin: 24mm 18mm; }
    body { background: var(--bg); color: var(--fg); font-family: system-ui, -apple-system, Segoe UI, Roboto, Ubuntu, Cantarell, Noto Sans, Helvetica, Arial, "Apple Color Emoji", "Segoe UI Emoji"; }
    .page { page-break-after: always; }
    .container { width: 100%; }
    h1, h2, h3 { margin: 0 0 6px; }
    .title { font-size: 28px; font-weight: 800; color: var(--fg); }
    .subtitle { color: var(--muted); margin-top: 4px; }
    .section { margin-top: 18px; padding: 12px; background: var(--card); border: 1px solid var(--border); border-radius: 8px; }
    .grid { display: grid; grid-template-columns: repeat(2, minmax(0, 1fr)); gap: 10px; }
    .metric { padding: 10px; border: 1px dashed var(--border); border-radius: 8px; background: white; }
    .metric .label { font-size: 12px; color: var(--muted); }
    .metric .value { font-size: 18px; font-weight: 700; margin-top: 2px; }
    .badge { display: inline-block; padding: 2px 8px; border-radius: 999px; font-size: 12px; background: #ede9fe; color: var(--accent); font-weight: 600; }
    .footer { margin-top: 24px; font-size: 12px; color: var(--muted); display: flex; justify-content: space-between; }
    .section h3 { color: var(--accent); }
    .muted { color: var(--muted); }
    .mono { font-family: ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, "Liberation Mono", monospace; }
  </style>
</head>
<body>
  <div class="container">
    ${body}
  </div>
</body>
</html>`;
}

function headerBlock(kind: string, input: ReportInput) {
  const now = input.generatedAt ?? new Date().toISOString();
  const period = input.period ? `<span class="badge">Period: ${escape(input.period)}</span>` : "";
  const by = input.generatedBy ? `<div class="muted">Generated by: ${escape(input.generatedBy)}</div>` : "";
  const summary = input.summary ? `<p class="muted" style="margin-top:8px">${escape(input.summary)}</p>` : "";
  return `
    <div class="page">
      <h1 class="title">${escape(input.title || `${kind} Report`)}</h1>
      <div class="subtitle">Type: ${escape(kind)} • <span class="mono">${escape(now)}</span></div>
      <div style="margin-top:8px">${period}</div>
      ${by}
      ${summary}
    </div>
  `;
}

function metricsGrid(metrics?: Array<{ label: string; value: string }>) {
  if (!metrics?.length) return "";
  const items = metrics
    .map(
      (m) => `
      <div class="metric">
        <div class="label">${escape(m.label)}</div>
        <div class="value">${escape(m.value)}</div>
      </div>`,
    )
    .join("");
  return `<div class="grid" style="margin-top:8px">${items}</div>`;
}

function sectionsBlock(sections?: Array<{ heading: string; body: string }>) {
  if (!sections?.length) return "";
  return sections
    .map(
      (s) => `
      <div class="section">
        <h3>${escape(s.heading)}</h3>
        <div>${s.body}</div>
      </div>`,
    )
    .join("");
}

export function buildReportHTML(template: ReportTemplateKey, input: ReportInput): string {
  const kind = template === "performance" ? "Performance" : template === "risk" ? "Risk" : "Compliance";

  const body = `
    ${headerBlock(kind, input)}
    ${metricsGrid(input.metrics)}
    ${sectionsBlock(input.sections)}
    <div class="footer">
      <span>Generated by GeFi</span>
      <span class="mono">Template: ${escape(template)}</span>
    </div>
  `;

  return baseHTML(input.title || `${kind} Report`, body);
}
```

```typescript name=server/services/storage.ts
import fs from "fs";
import fsp from "fs/promises";
import path from "path";

export const STORAGE_ROOT = process.env.STORAGE_ROOT || path.resolve(process.cwd(), "storage");
export const REPORTS_DIR = process.env.REPORTS_STORAGE_DIR || path.join(STORAGE_ROOT, "reports");
export const REPORTS_META_DIR = path.join(REPORTS_DIR, "metadata");

export async function ensureDirs() {
  await fsp.mkdir(STORAGE_ROOT, { recursive: true });
  await fsp.mkdir(REPORTS_DIR, { recursive: true });
  await fsp.mkdir(REPORTS_META_DIR, { recursive: true });
}

export async function saveBufferFile(outPath: string, data: Buffer) {
  await fsp.mkdir(path.dirname(outPath), { recursive: true });
  await fsp.writeFile(outPath, data);
  return outPath;
}

export async function writeJSON(outPath: string, data: any) {
  await fsp.mkdir(path.dirname(outPath), { recursive: true });
  await fsp.writeFile(outPath, JSON.stringify(data, null, 2), "utf-8");
}

export type StorageFile = {
  name: string;
  path: string;
  size: number;
  modifiedAt: string;
  ext: string;
  type: "Report" | "Dataset" | "AI Model" | "Other";
};

function classifyByExt(ext: string): StorageFile["type"] {
  const e = ext.toLowerCase();
  if (e === ".pdf") return "Report";
  if ([".csv", ".parquet", ".json"].includes(e)) return "Dataset";
  if ([".pkl", ".h5", ".pt", ".onnx", ".joblib"].includes(e)) return "AI Model";
  return "Other";
}

export async function listFilesRecursive(dir = STORAGE_ROOT): Promise<StorageFile[]> {
  const results: StorageFile[] = [];
  const walk = async (d: string) => {
    const entries = await fsp.readdir(d, { withFileTypes: true });
    for (const ent of entries) {
      const p = path.join(d, ent.name);
      if (ent.isDirectory()) {
        await walk(p);
      } else {
        const stat = await fsp.stat(p);
        const ext = path.extname(ent.name);
        results.push({
          name: ent.name,
          path: p,
          size: stat.size,
          modifiedAt: stat.mtime.toISOString(),
          ext,
          type: classifyByExt(ext),
        });
      }
    }
  };
  if (fs.existsSync(dir)) {
    await walk(dir);
  }
  return results;
}
```

```typescript name=server/services/reportGenerator.ts
import path from "path";
import { nanoid } from "nanoid";
import { buildReportHTML, ReportTemplateKey, ReportInput } from "../templates/reportTemplates";
import { ensureDirs, REPORTS_DIR, REPORTS_META_DIR, saveBufferFile, writeJSON } from "./storage";

// Lazy import puppeteer only when used
async function getPuppeteer() {
  // @ts-ignore
  const mod = await import("puppeteer");
  return mod.default || (mod as any);
}

export type ReportMetadata = {
  id: string;
  template: ReportTemplateKey;
  title: string;
  filename: string;
  absolutePath: string;
  size: number;
  createdAt: string;
  contentType: "application/pdf";
  status: "completed";
};

export async function generateReportPDF(template: ReportTemplateKey, input: ReportInput): Promise<ReportMetadata> {
  await ensureDirs();

  const id = nanoid(10);
  const title = input.title || (template === "performance" ? "Performance Report" : template === "risk" ? "Risk Report" : "Compliance Report");
  const filename = `report-${id}.pdf`;
  const outPath = path.join(REPORTS_DIR, filename);
  const html = buildReportHTML(template, {
    ...input,
    generatedAt: input.generatedAt || new Date().toISOString(),
  });

  const puppeteer = await getPuppeteer();
  const browser = await puppeteer.launch({
    headless: "new",
    args: ["--no-sandbox", "--disable-setuid-sandbox"],
  });

  try {
    const page = await browser.newPage();
    await page.setContent(html, { waitUntil: "networkidle0" });
    const pdfBuffer: Buffer = await page.pdf({
      printBackground: true,
      preferCSSPageSize: true,
      format: "A4",
      margin: { top: "16mm", bottom: "16mm", left: "14mm", right: "14mm" },
    });
    await saveBufferFile(outPath, pdfBuffer);
  } finally {
    await browser.close();
  }

  const size = (await (await import("fs/promises")).stat(outPath)).size;
  const metadata: ReportMetadata = {
    id,
    template,
    title,
    filename,
    absolutePath: outPath,
    size,
    createdAt: new Date().toISOString(),
    contentType: "application/pdf",
    status: "completed",
  };
  await writeJSON(path.join(REPORTS_META_DIR, `${id}.json`), metadata);
  return metadata;
}

export async function listReportMetadata(): Promise<ReportMetadata[]> {
  const fs = await import("fs/promises");
  const pathNode = await import("path");
  await ensureDirs();
  const entries = await fs.readdir(REPORTS_META_DIR).catch(() => []);
  const list: ReportMetadata[] = [];
  for (const f of entries) {
    if (!f.endsWith(".json")) continue;
    const p = pathNode.join(REPORTS_META_DIR, f);
    try {
      const raw = await fs.readFile(p, "utf-8");
      list.push(JSON.parse(raw));
    } catch {}
  }
  // newest first
  return list.sort((a, b) => (a.createdAt > b.createdAt ? -1 : 1));
}

export async function getReportById(id: string): Promise<ReportMetadata | null> {
  const fs = await import("fs/promises");
  const pathNode = await import("path");
  const p = pathNode.join(REPORTS_META_DIR, `${id}.json`);
  try {
    const raw = await fs.readFile(p, "utf-8");
    return JSON.parse(raw);
  } catch {
    return null;
  }
}
```

```typescript name=server/routes/reports.ts
import type { Express, Request, Response } from "express";
import path from "path";
import fs from "fs";
import { generateReportPDF, getReportById, listReportMetadata } from "../services/reportGenerator";
import type { ReportTemplateKey, ReportInput } from "../templates/reportTemplates";

export default function registerReports(app: Express) {
  const base = "/api/reports";

  // Generate report synchronously (fast enough for our standard templates).
  app.post(`${base}/generate`, async (req: Request, res: Response) => {
    try {
      const { template, input }: { template: ReportTemplateKey; input: ReportInput } = req.body || {};
      if (!template) {
        return res.status(400).json({ message: "template is required" });
      }
      const meta = await generateReportPDF(template, input || {});
      return res.json({
        id: meta.id,
        status: "completed",
        title: meta.title,
        downloadUrl: `${base}/${meta.id}/download`,
        metadata: meta,
      });
    } catch (err: any) {
      console.error("report generate failed:", err);
      res.status(500).json({ message: "Failed to generate report", error: String(err?.message || err) });
    }
  });

  // List all generated reports
  app.get(base, async (_req, res) => {
    const list = await listReportMetadata();
    res.json(list);
  });

  // Basic status endpoint
  app.get(`${base}/:id/status`, async (req, res) => {
    const meta = await getReportById(req.params.id);
    if (!meta) return res.status(404).json({ message: "Not found" });
    res.json({ id: meta.id, status: meta.status, metadata: meta });
  });

  // Download a PDF
  app.get(`${base}/:id/download`, async (req, res) => {
    const meta = await getReportById(req.params.id);
    if (!meta) return res.status(404).json({ message: "Not found" });

    if (!fs.existsSync(meta.absolutePath)) {
      return res.status(410).json({ message: "File no longer available" });
    }
    res.setHeader("Content-Type", meta.contentType);
    res.setHeader("Content-Disposition", `attachment; filename="${meta.filename}"`);
    fs.createReadStream(meta.absolutePath).pipe(res);
  });

  // Describe available templates
  app.get(`${base}/templates`, (_req, res) => {
    res.json([
      {
        key: "performance",
        label: "Performance Report",
        fields: ["title", "period", "generatedBy", "summary", "metrics", "sections"],
      },
      {
        key: "risk",
        label: "Risk Report",
        fields: ["title", "period", "generatedBy", "summary", "metrics", "sections"],
      },
      {
        key: "compliance",
        label: "Compliance Report",
        fields: ["title", "period", "generatedBy", "summary", "metrics", "sections"],
      },
    ]);
  });
}
```

```typescript name=server/routes/storage.ts
import type { Express, Request, Response } from "express";
import path from "path";
import fs from "fs";
import { listFilesRecursive, STORAGE_ROOT } from "../services/storage";

export default function registerStorageRoutes(app: Express) {
  const base = "/api/storage";

  app.get(`${base}/files`, async (_req: Request, res: Response) => {
    const files = await listFilesRecursive();
    // Convert absolute path to a relative label + add a simple download URL for reports
    const normalized = files.map((f) => {
      const rel = f.path.startsWith(STORAGE_ROOT) ? f.path.slice(STORAGE_ROOT.length + 1) : f.path;
      const downloadUrl = f.path.includes("/reports/") && f.path.endsWith(".pdf")
        ? `/api/reports/${path.basename(f.path).replace("report-", "").replace(".pdf", "")}/download`
        : null;
      return { ...f, relativePath: rel, downloadUrl };
    });
    res.json(normalized);
  });

  // Optional: stream any file by absolute path query (secured in real prod; here for demo/dev)
  app.get(`${base}/stream`, (req: Request, res: Response) => {
    const filePath = req.query.path as string;
    if (!filePath || !filePath.startsWith(STORAGE_ROOT)) {
      return res.status(400).json({ message: "Invalid path" });
    }
    if (!fs.existsSync(filePath)) return res.status(404).json({ message: "Not found" });
    const ext = path.extname(filePath).toLowerCase();
    const contentType = ext === ".pdf" ? "application/pdf" : "application/octet-stream";
    res.setHeader("Content-Type", contentType);
    fs.createReadStream(filePath).pipe(res);
  });
}
```

```typescript name=client/src/lib/reports.ts
import { apiRequest } from "@/lib/api-client";

export type GenerateOptions = {
  template: "performance" | "risk" | "compliance";
  input: {
    title?: string;
    period?: string;
    generatedBy?: string;
    summary?: string;
    metrics?: Array<{ label: string; value: string }>;
    sections?: Array<{ heading: string; body: string }>;
  };
};

export async function generateReport(opts: GenerateOptions) {
  return apiRequest("POST", "/api/reports/generate", opts);
}

// Example: attach to a “Export” button on any page
export async function exportAndDownload(opts: GenerateOptions) {
  const result: any = await generateReport(opts);
  const url = result?.downloadUrl;
  if (url) {
    window.open(url, "_blank");
  }
  return result;
}
```

```tsx name=client/src/components/reports/GenerateReportButton.tsx
  );
```

````markdown name=docs/Reports-System.md
# Reports System

This adds a simple, production-minded pipeline to generate printable PDFs from our report pages and store them under `storage/reports/`, with metadata and download endpoints.

## Endpoints

- `POST /api/reports/generate`
  - Body: `{ template: "performance"|"risk"|"compliance", input: { title, period, generatedBy, summary, metrics, sections } }`
  - Returns: `{ id, status, title, downloadUrl, metadata }`
- `GET /api/reports` – list previous reports (metadata)
- `GET /api/reports/:id/status` – basic status polling
- `GET /api/reports/:id/download` – download the PDF
- `GET /api/reports/templates` – available templates
- `GET /api/storage/files` – full storage listing (so PDFs appear in the Storage UI)

## Storage

- PDFs at: `storage/reports/report-<id>.pdf`
- Metadata at: `storage/reports/metadata/<id>.json`

## Developer setup

1) Install Puppeteer on the server:
   - `npm install puppeteer`

2) Mount routes in `server/index.ts`:
   ```ts
   import registerReports from "./routes/reports";
   import registerStorageRoutes from "./routes/storage";
   // ...
   registerReports(app);
   registerStorageRoutes(app);
   ```

3) Use the reusable button in pages:
   ```tsx
   import GenerateReportButton from "@/components/reports/GenerateReportButton";
   ```

## Notes

- Puppeteer runs headless; Replit requires `--no-sandbox` which is already configured.
- If you later want asynchronous jobs, swap the synchronous call in `POST /api/reports/generate` with a queue and a worker – the templates and storage layers can be reused unchanged.
````
