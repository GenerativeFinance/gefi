```typescript name=server/db/prismaClient.ts
// Thin Prisma client wrapper for server code to import.
// Ensure ENV var DATABASE_URL is set before running Prisma migrations or the app.
import { PrismaClient } from '@prisma/client';

declare global {
  // prevent multiple instances during hot-reload in development
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  var __prismaClient__: PrismaClient | undefined;
}

const prisma: PrismaClient =
  global.__prismaClient__ ||
  new PrismaClient({
    log: process.env.NODE_ENV === 'development' ? ['query', 'info', 'warn', 'error'] : ['warn', 'error'],
  });

if (process.env.NODE_ENV === 'development') {
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  global.__prismaClient__ = prisma;
}

export default prisma;
```

```typescript name=server/models/subscriptions.ts
/**
 * Prisma-backed subscriptions helper.
 * - Uses server/db/prismaClient.ts which exports a PrismaClient.
 * - Functions are typed and return plain JS objects.
 *
 * Usage:
 *  const subs = require('../models/subscriptions');
 *  await subs.createSubscription({ userId, modelId, price, currency, status, metadata });
 */

import prisma from "../db/prismaClient";
import { Decimal } from "@prisma/client/runtime";

type CreatePayload = {
  userId: string | number;
  modelId: number;
  price?: number | null;
  currency?: string | null;
  status?: string;
  metadata?: Record<string, any>;
};

export async function createSubscription(payload: CreatePayload) {
  const rec = await prisma.subscription.create({
    data: {
      userId: String(payload.userId),
      modelId: payload.modelId,
      // Prisma's Decimal type is used for numeric columns; prisma will coerce numbers automatically.
      price: payload.price !== undefined && payload.price !== null ? new Decimal(String(payload.price)) : null,
      currency: payload.currency ?? "USD",
      status: payload.status ?? "active",
      metadata: payload.metadata ?? null,
    },
  });
  return rec;
}

export async function findSubscriptionByUserAndModel(userId: string | number, modelId: number) {
  const row = await prisma.subscription.findFirst({
    where: {
      userId: String(userId),
      modelId,
    },
  });
  return row;
}

export async function listSubscriptionsByUser(userId: string | number) {
  const rows = await prisma.subscription.findMany({
    where: { userId: String(userId) },
    orderBy: { createdAt: "desc" },
  });
  return rows;
}

export async function updateSubscription(id: number, patch: Partial<Record<string, any>>) {
  // Only allow certain updatable fields to avoid overwriting createdAt etc.
  const data: any = {};
  if (patch.status !== undefined) data.status = patch.status;
  if (patch.price !== undefined) data.price = patch.price === null ? null : new Decimal(String(patch.price));
  if (patch.currency !== undefined) data.currency = patch.currency;
  if (patch.metadata !== undefined) data.metadata = patch.metadata;
  const updated = await prisma.subscription.update({
    where: { id },
    data,
  });
  return updated;
}

// Export default to keep compatibility with require(...) usages
export default {
  createSubscription,
  findSubscriptionByUserAndModel,
  listSubscriptionsByUser,
  updateSubscription,
};
```

```sql name=server/migrations/0001_create_subscriptions.sql
-- Optional raw SQL migration (Postgres). If you're using Prisma migrations, you can ignore this file.
CREATE TABLE IF NOT EXISTS subscriptions (
  id SERIAL PRIMARY KEY,
  user_id TEXT NOT NULL,
  model_id INTEGER NOT NULL,
  price NUMERIC(12,2),
  currency VARCHAR(8) DEFAULT 'USD',
  status VARCHAR(32) NOT NULL DEFAULT 'active',
  metadata JSONB,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_subscriptions_user ON subscriptions(user_id);
CREATE INDEX IF NOT EXISTS idx_subscriptions_model ON subscriptions(model_id);
```

Installation & migration steps (Prisma + Postgres)
1) Install packages
- If your project uses npm:
  - npm install @prisma/client
  - npm install -D prisma
- Or with yarn:
  - yarn add @prisma/client
  - yarn add -D prisma

2) Add DATABASE_URL
- Set environment variable DATABASE_URL to a Postgres connection string (for example in .env):
  - DATABASE_URL="postgresql://USER:PASSWORD@HOST:PORT/DATABASE?schema=public"

3) Generate Prisma client and run migration
- Initialize Prisma (if not already initialized):
  - npx prisma generate
- Create & run a migration (development):
  - npx prisma migrate dev --name init_subscriptions
  This will create the subscriptions table as per prisma/schema.prisma and generate the client.

4) Use the new helpers
- The existing server routes (server/routes/subscriptions.ts or server/routes/gefiRoutes.ts where you mount the subscriptions route) will call require('../models/subscriptions').default; that default now points to the Prisma-backed helpers.
- Ensure the new server/db/prismaClient.ts file is included in your project and the process has DATABASE_URL set when starting the server.

Notes & integrations
- Authentication: the route file uses req.user to determine the user ID. Make sure your auth middleware populates req.user (or adapt ensureAuth to map to your auth).
- Transactional operations: if you need to create related rows in other tables as part of subscription creation, use prisma.$transaction([...]) with the prisma client.
- Decimal type: Prisma maps NUMERIC to Decimal; the code uses Decimal where needed. Add import { Decimal } from '@prisma/client/runtime' if you need to use Decimal directly.

Testing locally (quick)
- Start a local Postgres instance (Docker quick run):
  docker run --rm -e POSTGRES_PASSWORD=pass -e POSTGRES_USER=user -e POSTGRES_DB=dev -p 5432:5432 -d postgres:15
- Set DATABASE_URL to postgresql://user:pass@localhost:5432/dev
- Run npx prisma migrate dev --name init_subscriptions
- Start server and call subscription endpoints as an authenticated user.
